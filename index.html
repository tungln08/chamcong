/** CHẤM CÔNG — API lưu dữ liệu. Dán vào Code.gs trong Apps Script. */ var TEN_SHEET = 'ChamCong'; var TZ = 'Asia/Ho_Chi_Minh'; function doGet(e) { var act = (e && e.parameter && e.parameter.action) || ''; var kq; try { if (act === 'cham') { kq = chamCong(e.parameter.ten, e.parameter.loai); } else if (act === 'lichsu') { kq = { ok: true, data: layTatCa().reverse() }; // mới nhất lên đầu } else if (act === 'homnay') { kq = { ok: true, homNay: layHomNay() }; } else { kq = { ok: false, loi: 'Sai action' }; } } catch (err) { kq = { ok: false, loi: String(err) }; } return ContentService .createTextOutput(JSON.stringify(kq)) .setMimeType(ContentService.MimeType.JSON); } function chamCong(ten, loai) { var sheet = laySheet(); var now = new Date(); var gio = Utilities.formatDate(now, TZ, 'HH:mm'); sheet.appendRow([ now, Utilities.formatDate(now, TZ, 'dd/MM/yyyy'), gio, loai, ten ]); // appendRow xếp hàng chờ, phải ép ghi xong rồi mới đọc lại được SpreadsheetApp.flush(); // trạng thái đọc từ sheet, rồi cộng thêm lượt vừa bấm cho chắc var h = layHomNay(); if (loai === 'Vào ca' && !h.vao) h.vao = gio; if (loai === 'Ra ca') h.ra = gio; return { ok: true, gio: gio, homNay: h }; } function layHomNay() { var hnay = Utilities.formatDate(new Date(), TZ, 'dd/MM/yyyy'); var v = null, r = null; layTatCa().forEach(function (x) { if (x.ngay !== hnay) return; if (x.loai === 'Vào ca' && !v) v = x.gio; if (x.loai === 'Ra ca') r = x.gio; }); return { vao: v, ra: r }; } function layTatCa() { var sheet = laySheet(); var last = sheet.getLastRow(); if (last < 2) return []; return sheet.getRange(2, 1, last - 1, 5).getValues() .filter(function (r) { return r[0] instanceof Date; }) .map(function (r) { return { ngay: Utilities.formatDate(r[0], TZ, 'dd/MM/yyyy'), gio: Utilities.formatDate(r[0], TZ, 'HH:mm'), loai: String(r[3]), ten: String(r[4]) }; }); } function laySheet() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName(TEN_SHEET); if (!sh) { sh = ss.insertSheet(TEN_SHEET); sh.appendRow(['Thời gian', 'Ngày', 'Giờ', 'Loại', 'Tên']); sh.setFrozenRows(1); } sh.getRange('B:C').setNumberFormat('@'); // giữ Ngày và Giờ ở dạng chữ return sh; }
