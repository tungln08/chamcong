/** CHẤM CÔNG — API lưu dữ liệu. Dán vào Code.gs trong Apps Script. */ var TEN_SHEET = 'ChamCong'; var TZ = 'Asia/Ho_Chi_Minh'; function doGet(e) { var act = (e && e.parameter && e.parameter.action) || ''; var kq; try { if (act === 'cham') { kq = chamCong(e.parameter.ten, e.parameter.loai); } else if (act === 'lichsu') { kq = { ok: true, data: layTatCa().reverse() }; // mới nhất lên đầu } else if (act === 'homnay') { kq = { ok: true, homNay: layHomNay() }; } else { kq = { ok: false, loi: 'Sai action' }; } } catch (err) { kq = { ok: false, loi: String(err) }; } return ContentService .createTextOutput(JSON.stringify(kq)) .setMimeType(ContentService.MimeType.JSON); } function chamCong(ten, loai) { var sheet = laySheet(); var now = new Date(); sheet.appendRow([ now, Utilities.formatDate(now, TZ, 'dd/MM/yyyy'), Utilities.formatDate(now, TZ, 'HH:mm'), loai, ten ]); return { ok: true, gio: Utilities.formatDate(now, TZ, 'HH:mm'), homNay: layHomNay() }; } function layHomNay() { var hnay = Utilities.formatDate(new Date(), TZ, 'dd/MM/yyyy'); var v = null, r = null; layTatCa().forEach(function (x) { if (x.ngay !== hnay) return; if (x.loai === 'Vào ca' && !v) v = x.gio; if (x.loai === 'Ra ca') r = x.gio; }); return { vao: v, ra: r }; } function layTatCa() { var sheet = laySheet(); var last = sheet.getLastRow(); if (last < 2) return []; return sheet.getRange(2, 1, last - 1, 5).getValues().map(function (r) { return { ngay: dinhDang(r[1], 'dd/MM/yyyy'), gio: dinhDang(r[2], 'HH:mm'), loai: String(r[3]), ten: String(r[4]) }; }); } /** Sheets hay tự đổi "01:32" thành kiểu Time (mốc 30/12/1899). Nếu ô là Date thì format lại, còn là chuỗi thì giữ nguyên. */ function dinhDang(v, mau) { return (v instanceof Date) ? Utilities.formatDate(v, TZ, mau) : String(v); } function laySheet() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName(TEN_SHEET); if (!sh) { sh = ss.insertSheet(TEN_SHEET); sh.appendRow(['Thời gian', 'Ngày', 'Giờ', 'Loại', 'Tên']); sh.setFrozenRows(1); sh.getRange('B:C').setNumberFormat('@'); // giữ Ngày và Giờ ở dạng chữ } return sh; }
